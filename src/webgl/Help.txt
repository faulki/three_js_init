// dans le constructor
this.bassHistory = [];
this.lastBeatTime = 0;
this.beatSettings = { sensitivity: 1.3, minEnergy: 0.3, cooldown: 200 };

onBeat(callback) {
    this.beatCallbacks.add(callback);
}

update() {
    if (!this.analyser || this.audio.paused) return;

    this.analyser.getByteFrequencyData(this.data);

    const bass = this.getBandEnergy(20, 150);

    this.bassHistory.push(bass);
    if (this.bassHistory.length > 43) this.bassHistory.shift(); // ~0.7 s à 60 fps

    const average = this.bassHistory.reduce((a, b) => a + b, 0) / this.bassHistory.length;
    const now = performance.now();
    const { sensitivity, minEnergy, cooldown } = this.beatSettings;

    const isBeat = bass > average * sensitivity && bass > minEnergy && now - this.lastBeatTime > cooldown;

    if (isBeat) {
        this.lastBeatTime = now;
        this.beatCallbacks.forEach((cb) => cb(bass));
    }
}

// nouvelle méthode
flash(energy = 1) {
    gsap.killTweensOf([this.spot, this.beam.material]);

    this.spot.intensity = this.flashIntensity * energy;
    this.beam.material.opacity = 0.15;

    gsap.to(this.spot, { intensity: 0, duration: 0.3, ease: 'power2.out' });
    gsap.to(this.beam.material, { opacity: 0, duration: 0.3, ease: 'power2.out' });
}
